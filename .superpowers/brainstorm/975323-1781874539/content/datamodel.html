— Key Resolver (tenant → global fallback) —
// ApiKeyResolver — called before every fetch Mono<Optional<String>> resolve( String vendorId, TenantId tenantId ) { return repo.findByVendorAndTenant(vendorId, tenantId) .switchIfEmpty( repo.findGlobal(vendorId) // tenant_id IS NULL ) .map(k -> Optional.of(k.apiKey())) .defaultIfEmpty(Optional.empty()); }
// Resolution rules at runtime: // // 1. Tenant-specific key found → use it // 2. No tenant key → fall back to global // 3. requiresApiKey() == false → skip resolver // 4. requiresApiKey() == true, no key found // → log WARN, skip connector for this run // // api_key never returned in GET responses // (masked: "••••••••••••7f3a")
